const express = require("express"); const bodyParser = require("body-parser"); const app = express(); app.use(bodyParser.json()); /* ========================= 🧠 "DATABASE" (MIDlertidig) ========================= */ let users = []; let chats = {}; /* ========================= 🌐 FRONTEND (HTML) ========================= */ app.get("/", (req,res)=>{ res.send(` AI Skole Norge

💬 Chatter

`); }); /* ========================= 🔐 AUTH ========================= */ app.post("/register",(req,res)=>{ let {username,password} = req.body; users.push({username,password}); res.json({success:true}); }); app.post("/login",(req,res)=>{ let {username,password} = req.body; let user = users.find(u=>u.username===username && u.password===password); if(user) res.json({success:true}); else res.json({success:false}); }); /* ========================= 💬 CHAT BACKEND ========================= */ app.get("/chats",(req,res)=>{ let user = req.query.user; res.json(chats[user] || []); }); app.post("/newchat",(req,res)=>{ let user = req.body.user; if(!chats[user]) chats[user]=[]; let chat = {id: Date.now(), messages:[]}; chats[user].push(chat); res.json(chat); }); app.get("/messages",(req,res)=>{ let chatId = parseInt(req.query.chat); for(let u in chats){ let chat = chats[u].find(c=>c.id===chatId); if(chat) return res.json(chat.messages); } res.json([]); }); /* ========================= 🤖 AI (ENKEL) ========================= */ app.post("/send",(req,res)=>{ let {chat,message} = req.body; for(let u in chats){ let c = chats[u].find(x=>x.id===chat); if(c){ c.messages.push({role:"Du", text:message}); // 🔥 AI SVAR (bytt med OpenAI senere) let reply = "AI svar: " + message; c.messages.push({role:"AI", text:reply}); return res.json({ok:true}); } } res.json({error:true}); }); app.listen(3000, ()=>console.log("🚀 Server: http://localhost:3000"));
